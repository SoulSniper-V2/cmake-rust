cmake_minimum_required(VERSION 4.1)
set(CMAKE_EXPERIMENTAL_EXPORT_PACKAGE_INFO "b80be207-778e-46ba-8080-b23bba22639e")
project(cps C)
add_library(a STATIC a.c)
add_library(b SHARED b.c)
add_library(m MODULE b.c)
add_library(o OBJECT a.c)
add_library(i INTERFACE)
add_executable(tool main.c)
set_target_properties(a b PROPERTIES DEBUG_POSTFIX d)
target_compile_definitions(b PUBLIC B=1 INTERFACE C)
target_link_libraries(a PUBLIC b PRIVATE m_dummy)
target_link_libraries(i INTERFACE b)
add_library(m_dummy INTERFACE)
install(TARGETS a b m o i tool m_dummy EXPORT es
  ARCHIVE DESTINATION lib/$<CONFIG> LIBRARY DESTINATION lib RUNTIME DESTINATION bin
  OBJECTS DESTINATION obj)
install(TARGETS b EXPORT rel LIBRARY DESTINATION lib CONFIGURATIONS Release)
install(EXPORT es DESTINATION lib/cmake CONFIGURATIONS Release Debug)
install(EXPORT rel DESTINATION lib/rel CONFIGURATIONS Release COMPONENT dev)
install(PACKAGE_INFO es EXPORT es VERSION 1.2.3 DEFAULT_TARGETS a)
install(PACKAGE_INFO rel EXPORT rel DESTINATION share/cps CONFIGURATIONS release COMPONENT dev)
export(EXPORT es NAMESPACE es::)
export(EXPORT es PACKAGE_INFO es2)

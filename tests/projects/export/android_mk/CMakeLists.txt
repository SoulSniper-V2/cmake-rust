cmake_minimum_required(VERSION 3.20)
project(amk C CXX)
add_library(sh SHARED a.c)
target_include_directories(sh PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/inc> $<INSTALL_INTERFACE:include/sh>)
target_compile_options(sh INTERFACE -fno-rtti)
target_compile_definitions(sh INTERFACE SHDEF=1)
add_library(st STATIC b.cpp)
target_link_libraries(st PUBLIC sh /opt/prebuilt/libz.so m)
target_link_options(st INTERFACE -Wl,--as-needed -Wl,-z,now)
export(TARGETS sh st NAMESPACE ns:: ANDROID_MK ${CMAKE_BINARY_DIR}/out/Android.mk)
export(TARGETS sh ANDROID_MK ${CMAKE_BINARY_DIR}/out/second.mk)
export(TARGETS st ANDROID_MK ${CMAKE_BINARY_DIR}/out/second.mk APPEND)
install(TARGETS sh st EXPORT e LIBRARY DESTINATION lib64/x ARCHIVE DESTINATION arc INCLUDES DESTINATION include/extra)
install(EXPORT_ANDROID_MK e NAMESPACE p:: DESTINATION share/ndk FILE e.mk COMPONENT dev)

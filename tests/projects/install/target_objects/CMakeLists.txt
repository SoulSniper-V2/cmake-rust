cmake_minimum_required(VERSION 3.30)
project(p C)
add_library(o OBJECT o.c)
file(GENERATE OUTPUT g$<CONFIG>.txt CONTENT "$<CONFIG> $<TARGET_OBJECTS:o>\n$<PATH:GET_FILENAME,$<TARGET_OBJECTS:o>>")
add_custom_command(OUTPUT x.txt COMMAND echo $<TARGET_OBJECTS:o> > x.txt DEPENDS $<TARGET_OBJECTS:o> VERBATIM)
add_custom_target(t ALL DEPENDS x.txt)
add_custom_target(t2 ALL COMMAND echo "$<JOIN:$<TARGET_OBJECTS:o>,|>" DEPENDS $<TARGET_OBJECTS:o>)
add_library(s STATIC o.c $<TARGET_OBJECTS:o>)
target_compile_definitions(s PRIVATE "X=$<TARGET_OBJECTS:o>")
target_link_options(s PRIVATE "$<TARGET_OBJECTS:o>")
add_executable(e m.c)
target_link_libraries(e PRIVATE o "$<TARGET_OBJECTS:o>")
target_link_options(e PRIVATE "-Wl,$<TARGET_OBJECTS:o>")
set_source_files_properties(m.c PROPERTIES COMPILE_FLAGS "-DY=$<TARGET_OBJECTS:o>" OBJECT_DEPENDS "")
add_test(NAME q COMMAND e $<TARGET_OBJECTS:o>)
enable_testing()
install(FILES $<TARGET_OBJECTS:o> DESTINATION x)
install(TARGETS o EXPORT x OBJECTS DESTINATION obj)
export(EXPORT x FILE x.cmake)
set_property(TARGET s PROPERTY INTERFACE_LINK_OPTIONS "$<TARGET_OBJECTS:o>")
install(CODE "message([[$<TARGET_OBJECTS:o>]])")

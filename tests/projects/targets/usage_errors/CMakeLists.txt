cmake_minimum_required(VERSION 4.1)
project(x NONE)
add_library(lib STATIC CMakeLists.txt)
add_library(iface INTERFACE)
add_library(imp INTERFACE IMPORTED)
add_library(alias ALIAS lib)
add_custom_target(util)
target_include_directories(missing PUBLIC x)
target_include_directories(lib x)
target_include_directories(lib)
target_include_directories(lib PUBLIC)
target_include_directories(alias PUBLIC x)
target_include_directories(imp PUBLIC x)
target_include_directories(imp INTERFACE imp-inc)
target_include_directories(iface PRIVATE x)
target_include_directories(iface INTERFACE rel "" /abs $<1:gen>)
target_include_directories(util PRIVATE x)
target_compile_definitions(lib PRIVATE -DA -D "" B=1 -DC=$<1:x>)
target_compile_definitions(lib PRIVATE)
target_compile_options(lib BEFORE)
target_compile_options(lib PRIVATE BEFORE -x)
target_link_options(lib PRIVATE SHELL:-a -b)
target_link_directories(lib PRIVATE rel /abs $<1:g>)
target_link_directories(iface PRIVATE rel)
target_sources(lib)
target_sources(lib PRIVATE)
target_sources(lib PRIVATE rel.c /abs/x.c $<1:g.c>)
target_sources(imp INTERFACE imported.c)
target_sources(lib PUBLIC FILE_SET)
target_sources(lib PUBLIC FILE_SET fs)
target_sources(lib PUBLIC FILE_SET fs TYPE BOGUS)
target_sources(lib PUBLIC FILE_SET HEADERS TYPE CXX_MODULES)
target_sources(lib PUBLIC FILE_SET CXX_MODULES FILES m.cppm)
target_sources(lib PRIVATE FILE_SET hs TYPE HEADERS FILES a.h)
target_sources(lib PUBLIC FILE_SET hs FILES b.h)
target_sources(lib PRIVATE FILE_SET hs FILES c.h)
target_sources(lib PRIVATE FILE_SET Bad TYPE HEADERS)
target_sources(lib PRIVATE FILE_SET _bad TYPE HEADERS)
target_sources(iface PRIVATE FILE_SET ih TYPE HEADERS)
target_sources(lib PRIVATE FILE_SET hs2 TYPE HEADERS BASE_DIRS /elsewhere FILES /tmp/outside.h)
target_precompile_headers(lib PRIVATE)
target_precompile_headers(lib REUSE_FROM)
target_precompile_headers(lib REUSE_FROM missing)
target_precompile_headers(lib REUSE_FROM iface)
target_precompile_headers(lib PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:a.h>" [[<b.h>]] [["c.h"]] rel.h)
target_compile_features(lib PRIVATE)
foreach(prop IN ITEMS INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES COMPILE_DEFINITIONS
    COMPILE_OPTIONS LINK_OPTIONS LINK_DIRECTORIES SOURCES PRECOMPILE_HEADERS HEADER_SETS
    INTERFACE_HEADER_SETS HEADER_SET_hs HEADER_DIRS_hs)
  get_property(value TARGET lib PROPERTY "${prop}")
  string(REPLACE "${CMAKE_CURRENT_SOURCE_DIR}" "<SOURCE>" value "${value}")
  message("lib ${prop}=[${value}]")
endforeach()
get_property(value TARGET iface PROPERTY INTERFACE_INCLUDE_DIRECTORIES)
string(REPLACE "${CMAKE_CURRENT_SOURCE_DIR}" "<SOURCE>" value "${value}")
message("iface INTERFACE_INCLUDE_DIRECTORIES=[${value}]")
get_property(value TARGET imp PROPERTY INTERFACE_INCLUDE_DIRECTORIES)
string(REPLACE "${CMAKE_CURRENT_SOURCE_DIR}" "<SOURCE>" value "${value}")
message("imp INTERFACE_INCLUDE_DIRECTORIES=[${value}]")

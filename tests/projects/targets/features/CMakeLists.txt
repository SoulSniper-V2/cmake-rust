cmake_minimum_required(VERSION 4.1)
project(x NONE)
add_library(lib INTERFACE)
add_library(imp INTERFACE IMPORTED)
target_compile_features(lib INTERFACE cxx_std_17)
target_compile_features(lib INTERFACE c_std_99 cxx_constexpr)
target_compile_features(lib INTERFACE bogus_feature)
target_compile_features(lib INTERFACE $<1:cxx_std_20>)
target_compile_features(imp INTERFACE cxx_std_11)
set(CMAKE_CXX_COMPILE_FEATURES cxx_std_98 cxx_std_11 cxx_constexpr)
set(CMAKE_CXX_STANDARD_COMPUTED_DEFAULT 98)
target_compile_features(lib INTERFACE cxx_constexpr cxx_std_11)
target_compile_features(lib INTERFACE cxx_std_23)
target_compile_features(lib INTERFACE cxx_lambdas)
target_compile_features(lib INTERFACE cuda_std_11)
get_property(f TARGET lib PROPERTY INTERFACE_COMPILE_FEATURES)
message("lib features=[${f}]")
get_property(f TARGET imp PROPERTY INTERFACE_COMPILE_FEATURES)
message("imp features=[${f}]")

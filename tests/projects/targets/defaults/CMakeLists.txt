cmake_minimum_required(VERSION 4.1)
project(x NONE)

set(CMAKE_BUILD_TYPE Debug)
set(CMAKE_POSITION_INDEPENDENT_CODE OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY app)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_DEBUG archive-debug)
set(CMAKE_DEBUG_POSTFIX _dbg)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_INSTALL_RPATH foo bar)
set(CMAKE_MSVC_RUNTIME_LIBRARY custom)
set(CMAKE_EXECUTABLE_ENABLE_EXPORTS ON)
set(CMAKE_MAP_IMPORTED_CONFIG_DEBUG Release)

add_library(obj OBJECT CMakeLists.txt)
add_executable(app WIN32 MACOSX_BUNDLE EXCLUDE_FROM_ALL CMakeLists.txt)
add_executable(tool IMPORTED GLOBAL)
add_library(sdk::stub UNKNOWN IMPORTED)
add_library(sdk::alias ALIAS sdk::stub)
set_property(TARGET obj app PROPERTY LINKER_LANGUAGE C)
set_property(TARGET sdk::stub PROPERTY IMPORTED_GLOBAL TRUE)

foreach(target IN ITEMS obj app tool sdk::stub sdk::alias)
  foreach(prop IN ITEMS NAME TYPE IMPORTED IMPORTED_GLOBAL ALIASED_TARGET ALIAS_GLOBAL
      EXCLUDE_FROM_ALL WIN32_EXECUTABLE MACOSX_BUNDLE ENABLE_EXPORTS
      POSITION_INDEPENDENT_CODE CXX_STANDARD CXX_EXTENSIONS CXX_VISIBILITY_PRESET
      RUNTIME_OUTPUT_DIRECTORY ARCHIVE_OUTPUT_DIRECTORY_DEBUG DEBUG_POSTFIX
      BUILD_RPATH_USE_ORIGIN INSTALL_RPATH SKIP_BUILD_RPATH MSVC_RUNTIME_LIBRARY
      MAP_IMPORTED_CONFIG_DEBUG PCH_WARN_INVALID UNITY_BUILD_BATCH_SIZE SYSTEM)
    get_target_property(value "${target}" "${prop}")
    message("${target}.${prop}=[${value}]")
  endforeach()
endforeach()

cmake_minimum_required(VERSION 3.28)
project(ks VERSION 1.2.3 LANGUAGES C CXX)
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo OUTPUT ipo_out LANGUAGES C CXX)
include(CheckPIESupported)
check_pie_supported(OUTPUT_VARIABLE pie_out LANGUAGES C CXX)
include(GenerateExportHeader)
include(CMakePackageConfigHelpers)
include(GNUInstallDirs)
include(CheckCXXSourceCompiles)
include(CheckSymbolExists)
include(CheckTypeSize)
include(CheckStructHasMember)
include(TestBigEndian)
test_big_endian(BIG)
check_type_size("long double" LD)
check_struct_has_member("struct stat" st_mtim sys/stat.h HAS_MTIM)
check_cxx_source_compiles("#include <optional>\nint main(){std::optional<int> o; return 0;}" HAS_OPT)
check_symbol_exists(strlcpy string.h HAS_STRLCPY)
find_package(Threads REQUIRED)
find_package(ZLIB)
find_package(PkgConfig)
if(PKG_CONFIG_FOUND)
  pkg_check_modules(ZPC IMPORTED_TARGET zlib)
endif()
set(CMAKE_CXX_COMPILER_LAUNCHER env)
add_library(lib SHARED lib.cpp)
generate_export_header(lib)
target_include_directories(lib PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(lib PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON INTERPROCEDURAL_OPTIMIZATION ${ipo} UNITY_BUILD ON)
target_precompile_headers(lib PRIVATE <vector> <string>)
target_link_libraries(lib PRIVATE Threads::Threads $<$<TARGET_EXISTS:ZLIB::ZLIB>:ZLIB::ZLIB>)
target_compile_features(lib PUBLIC cxx_std_20)
add_executable(app main.c)
set_target_properties(app PROPERTIES POSITION_INDEPENDENT_CODE ON C_LINKER_LAUNCHER env LINK_WHAT_YOU_USE ON)
target_link_libraries(app PRIVATE lib)
enable_testing()
add_test(NAME t1 COMMAND app)
set_tests_properties(t1 PROPERTIES LABELS "a;b" TIMEOUT 5 ENVIRONMENT_MODIFICATION "X=set:1")
install(TARGETS lib app EXPORT ksT RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/lib_export.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT ksT NAMESPACE ks:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ks)
configure_package_config_file(Config.cmake.in ksConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/ks)
write_basic_package_version_file(ksConfigVersion.cmake COMPATIBILITY SameMajorVersion)
export(EXPORT ksT NAMESPACE ks:: FILE ksTargets.cmake)
set(CPACK_PACKAGE_NAME ks)
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "x")
include(CPack)
include(CTest)

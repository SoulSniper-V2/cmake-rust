cmake_minimum_required(VERSION 3.16)
project(Imported CXX)
# The Qt tools come from imported targets, run once with -h.
foreach(tool moc uic rcc)
  add_executable(Qt5::${tool} IMPORTED)
  set_target_properties(Qt5::${tool} PROPERTIES
    IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/tools/${tool})
endforeach()
set(Qt5Core_VERSION_MAJOR 5)
set(Qt5Core_VERSION_MINOR 15)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOGEN_VERBOSE 2)
set(CMAKE_AUTOMOC_RELAXED_MODE ON)
set(CMAKE_AUTOGEN_PARALLEL bogus)
# CMP0071 and CMP0100 are not set by 3.16 (CMP0100) / NEW by 3.10 (CMP0071).
add_custom_command(OUTPUT gen.hh COMMAND ${CMAKE_COMMAND} -E touch gen.hh)
add_library(a STATIC a.cpp a.hh ${CMAKE_CURRENT_BINARY_DIR}/gen.hh a.qrc)
cmake_policy(SET CMP0071 OLD)
add_library(b STATIC a.cpp gen.hh)
target_link_libraries(a b)
target_link_libraries(b a)
set_property(TARGET a b PROPERTY AUTOGEN_ORIGIN_DEPENDS ON)
add_library(c SHARED a.cpp b.cpp)
set_property(TARGET c PROPERTY UNITY_BUILD ON)
set_property(SOURCE a.cpp b.cpp PROPERTY SKIP_PRECOMPILE_HEADERS ON)
target_precompile_headers(c PRIVATE <vector>)

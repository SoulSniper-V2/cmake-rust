cmake_minimum_required(VERSION 3.30)
project(modules_scan CXX)
set(CMAKE_CXX_MODULE_MAP_FORMAT gcc)
set(CMAKE_CXX_MODULE_MAP_FLAG "-mapper=<MODULE_MAP_FILE>")
set(CMAKE_CXX_SCANDEP_SOURCE "")
# The standard arrives through usage requirements or single features.
add_library(feat INTERFACE)
target_compile_features(feat INTERFACE cxx_std_20)
add_library(transitive INTERFACE)
target_link_libraries(transitive INTERFACE feat)
add_library(inherit STATIC a.cxx)
target_link_libraries(inherit PRIVATE transitive)
add_library(conf STATIC a.cxx)
target_link_libraries(conf PRIVATE $<$<CONFIG:Debug>:feat>)
add_library(old STATIC a.cxx)
set_property(TARGET old PROPERTY CXX_STANDARD 17)
set_property(TARGET old PROPERTY CXX_SCAN_FOR_MODULES ON)
add_library(prop STATIC a.cxx b.cxx)
set_property(TARGET prop PROPERTY CXX_STANDARD 23)
set_property(TARGET prop PROPERTY CXX_SCAN_FOR_MODULES OFF)
set_source_files_properties(b.cxx PROPERTIES CXX_SCAN_FOR_MODULES ON)
add_library(setoff STATIC c.cxx)
target_compile_features(setoff PRIVATE cxx_std_20)
set_property(TARGET setoff PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_sources(setoff PUBLIC FILE_SET CXX_MODULES FILES c.cxx)
add_library(unity STATIC a.cxx b.cxx c.cxx d.cxx)
target_compile_features(unity PRIVATE cxx_std_20)
set_source_files_properties(a.cxx d.cxx PROPERTIES CXX_SCAN_FOR_MODULES OFF)
set_property(TARGET unity PROPERTY UNITY_BUILD ON)
add_library(uplain STATIC a.cxx b.cxx)
set_property(TARGET uplain PROPERTY UNITY_BUILD ON)

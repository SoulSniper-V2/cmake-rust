cmake_minimum_required(VERSION 3.30)
project(unity_nmc C CXX)
set(CMAKE_UNITY_BUILD ON)
file(WRITE ${CMAKE_BINARY_DIR}/gen/g.c "int g;\n")
add_library(conf STATIC a.c $<$<CONFIG:Debug>:d.c> $<$<NOT:$<CONFIG:Debug>>:r.c> b.cxx ${CMAKE_BINARY_DIR}/gen/g.c)
set_target_properties(conf PROPERTIES UNITY_BUILD_UNIQUE_ID UID
  UNITY_BUILD_CODE_BEFORE_INCLUDE "// before" UNITY_BUILD_CODE_AFTER_INCLUDE "// after")
add_library(batch STATIC a.c $<$<CONFIG:Release>:r.c> d.c)
set_target_properties(batch PROPERTIES UNITY_BUILD_BATCH_SIZE 2 UNITY_BUILD_RELOCATABLE ON)
add_library(group STATIC a.c $<$<CONFIG:Release>:r.c> d.c)
set_target_properties(group PROPERTIES UNITY_BUILD_MODE GROUP)
set_source_files_properties(a.c r.c PROPERTIES UNITY_GROUP one)
target_precompile_headers(group PRIVATE "$<$<CONFIG:Debug>:<stdio.h>>" "$<$<CONFIG:Release>:<stdlib.h>>")
add_subdirectory(sub)

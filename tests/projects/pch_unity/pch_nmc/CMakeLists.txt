cmake_minimum_required(VERSION 3.30)
project(pch_nmc C CXX)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release;Extra" CACHE STRING "" FORCE)
add_library(owner STATIC a.c b.cxx)
target_precompile_headers(owner PUBLIC "$<$<COMPILE_LANGUAGE:C>:<stdio.h>>" "$<$<CONFIG:Release>:<stddef.h>>" own.h)
set_source_files_properties(skip.c PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
add_library(reuse SHARED a.c b.cxx skip.c)
target_precompile_headers(reuse REUSE_FROM owner)
# C sources in one configuration only.
add_executable(only main.cxx $<$<CONFIG:Debug>:a.c>)
target_precompile_headers(only PRIVATE <string.h>)
add_executable(late main.cxx $<$<CONFIG:Extra>:a.c>)
target_precompile_headers(late PRIVATE <string.h>)
add_library(consumer OBJECT a.c)
target_link_libraries(consumer PRIVATE owner)
add_library(off STATIC a.c)
target_precompile_headers(off PRIVATE <stdio.h>)
set_target_properties(off PROPERTIES DISABLE_PRECOMPILE_HEADERS ON)

cmake_minimum_required(VERSION 3.30)
project(limit C CXX)
set(CMAKE_INSTALL_PREFIX /prefix)
# The link lines exceed the command line length limit.
string(REPEAT "x" 80 long)
set(sources)
foreach(i RANGE 1000)
  file(WRITE ${CMAKE_BINARY_DIR}/g/${long}/${long}${i}.c "int f${i};")
  list(APPEND sources ${CMAKE_BINARY_DIR}/g/${long}/${long}${i}.c)
endforeach()
add_library(s STATIC ${sources})
add_executable(e main.cxx ${sources})
add_library(m MODULE ${sources})
set(CMAKE_C_RESPONSE_FILE_LINK_FLAG "-rsp=")
set(CMAKE_C_USE_RESPONSE_FILE_FOR_LIBRARIES OFF)
add_library(small SHARED small.c)

cmake_minimum_required(VERSION 3.30)
project(rsp C CXX)
set(CMAKE_NINJA_FORCE_RESPONSE_FILE 1)
set(CMAKE_INSTALL_PREFIX /prefix)
add_library(s STATIC a.c)
add_library(sh SHARED b.cxx)
target_compile_definitions(sh PRIVATE X=1)
target_include_directories(sh PRIVATE inc)
add_library(o OBJECT a.c)
add_executable(e m.c $<TARGET_OBJECTS:o>)
target_link_libraries(e PRIVATE s sh)
target_precompile_headers(e PRIVATE <stdio.h>)

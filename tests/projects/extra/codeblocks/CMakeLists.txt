cmake_minimum_required(VERSION 3.20)
project(Extra C CXX)
add_compile_definitions(TOP_DEF=1)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/inc)
add_library(core STATIC core.c)
target_compile_definitions(core PUBLIC CORE_API=1 PRIVATE CORE_IMPL)
target_include_directories(core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core_inc)
target_compile_options(core PRIVATE -Wall)
add_library(shared SHARED shared.cpp)
target_link_libraries(shared PRIVATE core)
add_custom_command(OUTPUT gen.c COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_SOURCE_DIR}/gen.c.in gen.c
  DEPENDS gen.c.in COMMENT "generating gen.c")
add_executable(app main.cpp gen.c inc/app.h)
target_link_libraries(app PRIVATE shared core)
add_custom_target(docs COMMAND ${CMAKE_COMMAND} -E echo docs SOURCES README.txt)
add_subdirectory(sub)
enable_testing()
add_test(NAME app COMMAND app)
install(TARGETS app shared core)

cmake_minimum_required(VERSION 4.1)
project(relative_directories_error C)
add_library(i INTERFACE)
set_target_properties(i PROPERTIES INTERFACE_LINK_DIRECTORIES "rel1;rel2")
add_executable(x a.c)
set_property(TARGET x PROPERTY LINK_DIRECTORIES /ok p1 p2)
target_link_libraries(x i)
add_executable(y a.c)
target_link_libraries(y i)

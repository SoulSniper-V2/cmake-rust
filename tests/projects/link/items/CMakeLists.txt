cmake_minimum_required(VERSION 4.1)
project(items C CXX Fortran)
add_library(f STATIC f.f90)
add_library(x STATIC x.cxx)
add_executable(a a.c)
set_target_properties(a PROPERTIES LINKER_LANGUAGE C)
target_link_libraries(a x f)
add_executable(b a.c x.cxx f.f90)
set_target_properties(b PROPERTIES LINKER_LANGUAGE C)
# static mode
file(WRITE ${CMAKE_BINARY_DIR}/d1/libq.a "")
file(WRITE ${CMAKE_BINARY_DIR}/d1/libr.so "")
file(WRITE ${CMAKE_BINARY_DIR}/d2/libr.so "x")
file(WRITE ${CMAKE_BINARY_DIR}/d2/libw.so "")
file(WRITE ${CMAKE_BINARY_DIR}/d1/libw.so "y")
add_executable(c a.c)
target_link_libraries(c libfoo.a ${CMAKE_BINARY_DIR}/d1/libq.a bar libz.so ${CMAKE_BINARY_DIR}/d1/libr.so ${CMAKE_BINARY_DIR}/d2/libw.so ${CMAKE_BINARY_DIR}/d1)
add_library(s1 STATIC x.cxx)
add_executable(d a.c)
target_link_libraries(d /usr/lib/x86_64-linux-gnu/libm.so)
target_link_directories(d PRIVATE ${CMAKE_BINARY_DIR}/d3)
file(WRITE ${CMAKE_BINARY_DIR}/d3/libm.so "")
file(WRITE ${CMAKE_BINARY_DIR}/d3/libm.so.6 "")
target_link_libraries(d -la LINKER:-z,now "$<LINK_GROUP:RESCAN,f,x>" /nonex/libt.a.1 libu.so.1.2 foo.so)
set_target_properties(d PROPERTIES LINK_SEARCH_START_STATIC ON)
set_target_properties(c PROPERTIES LINK_SEARCH_END_STATIC ON)
add_executable(e a.c)
set_target_properties(e PROPERTIES LINK_SEARCH_END_STATIC ON)
target_link_libraries(e "$<LINK_LIBRARY:WHOLE_ARCHIVE,f,x,zz>" /usr/lib/x86_64-linux-gnu/libm.a /lib/x86_64-linux-gnu/libz.so.1)
add_library(imp SHARED IMPORTED)
set_target_properties(imp PROPERTIES IMPORTED_LOCATION /opt/i/libimp.so IMPORTED_NO_SONAME ON)
add_library(un UNKNOWN IMPORTED)
set_target_properties(un PROPERTIES IMPORTED_LOCATION /opt/u/libun.so.4)
add_library(m MODULE a.c)
add_executable(host a.c)
set_target_properties(host PROPERTIES ENABLE_EXPORTS ON)
target_link_libraries(m host imp un)
add_library(sh SHARED x.cxx)
target_link_libraries(sh PRIVATE un imp e1)
set_target_properties(sh PROPERTIES VERSION 1.2 SOVERSION 1)
target_link_libraries(a sh)
add_executable(g a.c)
target_link_libraries(g imp sh x)

cmake_minimum_required(VERSION 3.20)
project(isp C CXX ISPC)
add_library(k STATIC k.ispc main.c)
set_property(TARGET k PROPERTY ISPC_INSTRUCTION_SETS sse2-i32x4 avx1-i32x8 avx2-i32x8)
add_library(obj OBJECT o.ispc)
set_property(TARGET obj PROPERTY ISPC_HEADER_DIRECTORY hdr)
set_property(TARGET obj PROPERTY POSITION_INDEPENDENT_CODE ON)
add_library(sh SHARED s.cpp $<TARGET_OBJECTS:obj>)
target_link_libraries(sh PRIVATE k)
add_executable(app app.c)
target_link_libraries(app PRIVATE sh k)
set_property(TARGET k PROPERTY ISPC_HEADER_SUFFIX .hh)

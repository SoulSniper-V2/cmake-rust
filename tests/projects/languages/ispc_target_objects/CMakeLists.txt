cmake_minimum_required(VERSION 3.20)
project(isp2 C CXX ISPC)
add_library(kobj OBJECT k.ispc)
set_property(TARGET kobj PROPERTY ISPC_INSTRUCTION_SETS sse2-i32x4 avx2-i32x8)
set_property(TARGET kobj PROPERTY POSITION_INDEPENDENT_CODE ON)
add_library(st STATIC sub/o.ispc)
add_executable(app app.cpp $<TARGET_OBJECTS:kobj>)
target_link_libraries(app PRIVATE st kobj)
file(GENERATE OUTPUT objs.txt CONTENT "$<JOIN:$<TARGET_OBJECTS:kobj>,\n>\n")
install(TARGETS kobj EXPORT e OBJECTS DESTINATION obj)
install(TARGETS st app ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(EXPORT e DESTINATION share/e)
export(EXPORT e FILE e-build.cmake)

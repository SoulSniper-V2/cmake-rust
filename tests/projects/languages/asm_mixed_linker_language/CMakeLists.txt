cmake_minimum_required(VERSION 3.25)
project(asm LANGUAGES C ASM ASM-ATT)
add_library(a STATIC a.s b.S)
set_property(TARGET a PROPERTY LINKER_LANGUAGE C)
add_library(att STATIC c.asm)
set_source_files_properties(c.asm PROPERTIES LANGUAGE ASM-ATT)
target_compile_options(a PRIVATE $<$<COMPILE_LANGUAGE:ASM>:-Wa,--noexecstack>)
add_executable(e main.c)
target_link_libraries(e PRIVATE a att)

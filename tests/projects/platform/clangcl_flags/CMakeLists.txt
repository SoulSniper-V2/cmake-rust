cmake_minimum_required(VERSION 3.29)
project(MsvcFlags C CXX)
add_library(st STATIC a.c)
set_target_properties(st PROPERTIES STATIC_LIBRARY_OPTIONS "/NODEFAULTLIB" MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(app WIN32 main.cxx)
target_link_libraries(app PRIVATE st)
target_include_directories(app PRIVATE /opt/inc)
target_include_directories(app SYSTEM PRIVATE /opt/sys)
set_target_properties(app PROPERTIES MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase COMPILE_WARNING_AS_ERROR ON LINK_WARNING_AS_ERROR ON)
add_executable(con main.cxx)
set_target_properties(con PROPERTIES INTERPROCEDURAL_OPTIMIZATION ON MSVC_RUNTIME_CHECKS "StackFrameErrorCheck;UninitializedVariable")

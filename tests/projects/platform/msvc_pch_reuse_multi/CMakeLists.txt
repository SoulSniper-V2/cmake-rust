cmake_minimum_required(VERSION 3.29)
project(PdbReuse C CXX)
add_library(base STATIC a.cxx)
target_precompile_headers(base PRIVATE <vector>)
set_target_properties(base PROPERTIES MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
add_library(user STATIC b.cxx)
set_target_properties(user PROPERTIES PRECOMPILE_HEADERS_REUSE_FROM base PREFIX lib MSVC_DEBUG_INFORMATION_FORMAT EditAndContinue)
add_executable(app main.cxx)
set_target_properties(app PROPERTIES PRECOMPILE_HEADERS_REUSE_FROM base)
target_link_libraries(app PRIVATE user)

cmake_minimum_required(VERSION 4.1)
project(initial_cache NONE)
foreach(property IN ITEMS COMPILE_OPTIONS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS
    CMAKE_CONFIGURE_DEPENDS BUILDSYSTEM_TARGETS MINE)
  get_property(value DIRECTORY PROPERTY ${property})
  message(STATUS "${property}=${value}")
endforeach()
get_property(value GLOBAL PROPERTY FROM_PRELOAD)
get_property(defined GLOBAL PROPERTY DEFINED_IN_PRELOAD DEFINED)
message(STATUS "global=${value} defined=${defined} env=$ENV{FROM_PRELOAD}")
message(STATUS "FIRST=${FIRST} BOTH=${BOTH} LAST=${LAST} KEPT=${KEPT} normal=${NORMAL}")
if(COMMAND from_preload)
  message(STATUS "command kept")
endif()
# A copy must not consume a neighbouring <output>.tmp file.
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/copy.h.tmp "x")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/copy.h.tmp ${CMAKE_CURRENT_BINARY_DIR}/copy.h COPYONLY)
if(EXISTS ${CMAKE_CURRENT_BINARY_DIR}/copy.h.tmp)
  message(STATUS "copy.h.tmp kept")
endif()
file(CONFIGURE OUTPUT copy.h CONTENT "y")
if(EXISTS ${CMAKE_CURRENT_BINARY_DIR}/copy.h.tmp)
  message(STATUS "copy.h.tmp kept")
endif()

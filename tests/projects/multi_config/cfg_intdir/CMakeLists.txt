cmake_minimum_required(VERSION 4.1)
project(cfg_intdir C)
add_library(a STATIC a.c)
set_target_properties(a PROPERTIES ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${CMAKE_CFG_INTDIR}/lib)
add_library(s SHARED a.c)
set_target_properties(s PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${CMAKE_CFG_INTDIR}/slib SOVERSION 3)
add_executable(x a.c)
target_link_libraries(x a s -Wl,${CMAKE_CFG_INTDIR}/q "${CMAKE_CFG_INTDIR}/lib/foo bar.so" -L${CMAKE_CFG_INTDIR})
target_link_directories(x PRIVATE ${CMAKE_CFG_INTDIR}/z)
target_compile_definitions(x PRIVATE Q="${CMAKE_CFG_INTDIR}")
target_include_directories(x PRIVATE ${CMAKE_CFG_INTDIR}/i)
target_link_options(x PRIVATE -Wl,${CMAKE_CFG_INTDIR}/o)
target_compile_options(x PRIVATE -O${CMAKE_CFG_INTDIR})
set_target_properties(x PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/${CMAKE_CFG_INTDIR}/bin BUILD_RPATH "${CMAKE_CFG_INTDIR}/rp;/o;/t" LINK_FLAGS -p${CMAKE_CFG_INTDIR} COMPILE_FLAGS -k${CMAKE_CFG_INTDIR})
add_custom_command(OUTPUT ${CMAKE_CFG_INTDIR}/o.txt COMMAND ${CMAKE_CFG_INTDIR}/tool ${CMAKE_CFG_INTDIR}/arg x${CMAKE_CFG_INTDIR} WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_CFG_INTDIR} DEPENDS ${CMAKE_CFG_INTDIR}/i.txt COMMENT "c ${CMAKE_CFG_INTDIR}")
add_custom_command(OUTPUT v.txt COMMAND ${CMAKE_CFG_INTDIR}/tool ${CMAKE_CFG_INTDIR}/arg x x${CMAKE_CFG_INTDIR} WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_CFG_INTDIR} VERBATIM)
add_custom_target(t ALL COMMAND ${CMAKE_CFG_INTDIR}/tool "a ${CMAKE_CFG_INTDIR}" DEPENDS ${CMAKE_CFG_INTDIR}/o.txt v.txt)
add_custom_command(TARGET x POST_BUILD COMMAND ${CMAKE_CFG_INTDIR}/post ${CMAKE_CFG_INTDIR} VERBATIM)
install(TARGETS x s a)

cmake_minimum_required(VERSION 3.28)
project(CxxModulesCross CXX)
add_library(mod STATIC)
target_sources(mod PUBLIC FILE_SET fs TYPE CXX_MODULES FILES mod.cxx)
target_compile_features(mod PUBLIC cxx_std_20)
add_library(user STATIC user.cxx)
target_link_libraries(user PRIVATE mod)
add_executable(app main.cxx)
target_link_libraries(app PRIVATE user mod)

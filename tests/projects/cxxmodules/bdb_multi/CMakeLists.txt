cmake_minimum_required(VERSION 3.28)
project(CxxModulesBuildDatabase CXX)
set(CMAKE_EXPERIMENTAL_EXPORT_BUILD_DATABASE "73194a1d-c0b5-41b9-9190-a4512925e192")

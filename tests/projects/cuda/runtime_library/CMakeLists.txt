cmake_minimum_required(VERSION 3.26)
project(runtime_library LANGUAGES CUDA)
find_package(CUDAToolkit REQUIRED)
add_executable(runtime_static main.cu)
set_target_properties(runtime_static PROPERTIES CUDA_RUNTIME_LIBRARY Static)
add_executable(runtime_shared main.cu)
set_target_properties(runtime_shared PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
add_executable(runtime_none main.cu)
set_target_properties(runtime_none PROPERTIES CUDA_RUNTIME_LIBRARY None)
target_link_libraries(runtime_none PRIVATE CUDA::cudart)

add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/a.txt
  COMMAND gen a.txt
  COMMAND ${CMAKE_BINARY_DIR}/bin/gen b
  COMMAND ${CMAKE_CURRENT_BINARY_DIR}/local.sh
  COMMAND ${CMAKE_CURRENT_BINARY_DIR}/./dot.sh
  COMMAND ${CMAKE_COMMAND} -E touch a.txt
  DEPENDS gen)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/b.txt
  COMMAND gen b.txt
  COMMAND ${CMAKE_CURRENT_BINARY_DIR}/local.sh
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
add_custom_target(use ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/a.txt ${CMAKE_CURRENT_BINARY_DIR}/b.txt)

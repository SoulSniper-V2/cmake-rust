cmake_minimum_required(VERSION 4.1)
project(CTestSelection NONE)
enable_testing()
foreach(index RANGE 1 12)
  add_test(NAME "case-${index}" COMMAND /bin/echo "Test ${index}")
  math(EXPR parity "${index} % 2")
  set_tests_properties("case-${index}" PROPERTIES LABELS "common;parity-${parity}")
endforeach()
add_subdirectory(nested)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/names.txt" "case-4\ncase-10\nnested\n")

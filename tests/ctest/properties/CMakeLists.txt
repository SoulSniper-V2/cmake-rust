cmake_minimum_required(VERSION 4.1)
project(CTestProperties NONE)
enable_testing()
function(shell name script)
  add_test(NAME "${name}" COMMAND /bin/sh -c "${script}")
endfunction()
shell(pass "printf 'all good\\n'")
shell(fail "echo 'bad news'; exit 3")
shell(will-fail "exit 2")
set_tests_properties(will-fail PROPERTIES WILL_FAIL YES)
shell(required-regex "echo expected; exit 2")
set_tests_properties(required-regex PROPERTIES PASS_REGULAR_EXPRESSION "expected;alternate")
shell(missing-regex "echo unrelated")
set_tests_properties(missing-regex PROPERTIES PASS_REGULAR_EXPRESSION "expected;alternate")
shell(forbidden-regex "echo forbidden")
set_tests_properties(forbidden-regex PROPERTIES FAIL_REGULAR_EXPRESSION "forbidden")
shell(skipped-regex "echo unavailable; exit 5")
set_tests_properties(skipped-regex PROPERTIES SKIP_REGULAR_EXPRESSION "unavailable")
shell(skipped-code "exit 77")
set_tests_properties(skipped-code PROPERTIES SKIP_RETURN_CODE 77)
shell(disabled "exit 9")
set_tests_properties(disabled PROPERTIES DISABLED TRUE)
shell(environment "test \"$TEST_VALUE\" = original-world; printf '%s\\n' \"$TEST_VALUE\"")
set_tests_properties(environment PROPERTIES ENVIRONMENT "TEST_VALUE=original"
  ENVIRONMENT_MODIFICATION "TEST_VALUE=set:replace;TEST_VALUE=reset:;TEST_VALUE=string_append:-world")
shell(work-dir "test -f marker; pwd")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/work")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/work/marker" "")
set_tests_properties(work-dir PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/work" REQUIRED_FILES marker)
shell(missing-file "exit 1")
set_tests_properties(missing-file PROPERTIES REQUIRED_FILES missing)
add_test(NAME missing-executable COMMAND test_fixture_no_such_executable)
shell(timeout "echo waiting; sleep 4")
set_tests_properties(timeout PROPERTIES TIMEOUT 0.08)
shell(timeout-after-match "echo start; sleep 4")
set_tests_properties(timeout-after-match PROPERTIES TIMEOUT_AFTER_MATCH "0.08;start")
shell(timeout-signal "trap 'echo intercepted; exit 0' USR1; while :; do sleep 0.01; done")
set_tests_properties(timeout-signal PROPERTIES TIMEOUT 0.1 TIMEOUT_SIGNAL_NAME SIGUSR1 TIMEOUT_SIGNAL_GRACE_PERIOD 0.3)
shell(no-timeout "sleep 0.06")
set_tests_properties(no-timeout PROPERTIES TIMEOUT 0)
shell(global-timeout "sleep 1")
shell(with-properties "echo '<CTestLabel>dynamic</CTestLabel>'; echo 'text & more'")
set_tests_properties(with-properties PROPERTIES LABELS "x;y" PROCESSORS 2 COST 15
  ATTACHED_FILES "${CMAKE_CURRENT_BINARY_DIR}/work/marker" MEASUREMENT "score=three"
  CUSTOM_TEST_PROPERTY literal)
shell(depender "test -f stamp")
shell(dependency "touch stamp")
set_tests_properties(depender PROPERTIES DEPENDS dependency COST 20 RUN_SERIAL ON RESOURCE_LOCK data)
shell(eventual-pass "test -f retry && exit 0; touch retry; exit 1")
shell(eventual-fail "test -f retry && exit 1; touch retry")
shell(retry-timeout "test -f retry && exit 0; touch retry; sleep 1")
set_tests_properties(retry-timeout PROPERTIES TIMEOUT 0.05)

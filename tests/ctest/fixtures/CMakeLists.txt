cmake_minimum_required(VERSION 4.1)
project(CTestFixtures NONE)
enable_testing()
add_test(NAME consumer COMMAND /bin/sh -c "test -f ready")
set_tests_properties(consumer PROPERTIES FIXTURES_REQUIRED store)
add_test(NAME cleanup COMMAND /bin/rm -f ready)
set_tests_properties(cleanup PROPERTIES FIXTURES_CLEANUP store)
add_test(NAME setup COMMAND /usr/bin/touch ready)
set_tests_properties(setup PROPERTIES FIXTURES_SETUP store)
add_test(NAME broken-consumer COMMAND /bin/true)
set_tests_properties(broken-consumer PROPERTIES FIXTURES_REQUIRED broken)
add_test(NAME broken-cleanup COMMAND /bin/echo done)
set_tests_properties(broken-cleanup PROPERTIES FIXTURES_CLEANUP broken)
add_test(NAME broken-setup COMMAND /bin/false)
set_tests_properties(broken-setup PROPERTIES FIXTURES_SETUP broken)

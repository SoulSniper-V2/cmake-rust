cmake_minimum_required(VERSION 4.1)
project(CTestResources NONE)
enable_testing()
add_test(NAME allocated COMMAND /bin/sh -c "env | grep '^CTEST_RESOURCE_GROUP' | sort")
set_tests_properties(allocated PROPERTIES RESOURCE_GROUPS "2,gpu:2;gpu:1,board:1")
add_test(NAME missing-type COMMAND /bin/true)
set_tests_properties(missing-type PROPERTIES RESOURCE_GROUPS "other:1")
add_test(NAME exhausted COMMAND /bin/true)
set_tests_properties(exhausted PROPERTIES RESOURCE_GROUPS "gpu:99")
configure_file(resources.json resources.json COPYONLY)

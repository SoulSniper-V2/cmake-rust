cmake_minimum_required(VERSION 3.16)
project(BuildAndTestNested C)
set(TEST_RESULT 0 CACHE STRING "test exit code")
option(BREAK_BUILD "break compilation" OFF)
add_executable(bt main.c)
target_compile_definitions(bt PRIVATE TEST_RESULT=${TEST_RESULT})
if(BREAK_BUILD)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/broken.c" "this is not C\n")
  target_sources(bt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/broken.c")
endif()

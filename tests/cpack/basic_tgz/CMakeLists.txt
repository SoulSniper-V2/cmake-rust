cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" "nested\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/emptydir")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION share/probe PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" DESTINATION share/probe/nested)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/emptydir" DESTINATION share/probe)
include(CPack)

cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/p1.txt" "one\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/p2.txt" "two\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/s1.cmake" "file(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}/share/probe\" TYPE FILE FILES \"${CMAKE_CURRENT_BINARY_DIR}/p1.txt\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/s2.cmake" "file(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}/share/probe/sub\" TYPE FILE FILES \"${CMAKE_CURRENT_BINARY_DIR}/p2.txt\")\n")
set(CPACK_INSTALL_SCRIPTS "${CMAKE_CURRENT_BINARY_DIR}/s1.cmake;${CMAKE_CURRENT_BINARY_DIR}/s2.cmake")
set(CPACK_INSTALL_CMAKE_PROJECTS "")
include(CPack)

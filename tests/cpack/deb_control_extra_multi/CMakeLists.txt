cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/postinst" "#!/bin/sh\nexit 0\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/postrm" "#!/bin/sh\nexit 0\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/conffiles" "/etc/probe-a.conf\n/etc/probe-b.conf\n")
set(CPACK_PACKAGE_CONTACT "Acme Maintainer <acme@example.com>")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_BINARY_DIR}/postinst;${CMAKE_CURRENT_BINARY_DIR}/postrm;${CMAKE_CURRENT_BINARY_DIR}/conffiles")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION share/probe)
include(CPack)

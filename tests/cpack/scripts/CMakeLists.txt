cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" "nested\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/emptydir")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pre.cmake" "message(STATUS \"pre-running\")\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/post.cmake" "message(STATUS \"post-running\")\n")
set(CPACK_PRE_BUILD_SCRIPTS "${CMAKE_CURRENT_BINARY_DIR}/pre.cmake")
set(CPACK_POST_BUILD_SCRIPTS "${CMAKE_CURRENT_BINARY_DIR}/post.cmake")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION share/probe)
include(CPack)

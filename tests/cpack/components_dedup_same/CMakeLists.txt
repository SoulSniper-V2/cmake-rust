cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" "nested\n")
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
set(CPACK_COMPONENTS_IGNORE_GROUPS ON)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION same COMPONENT one)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION same COMPONENT two)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" DESTINATION other COMPONENT two)
include(CPack)

cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" "nested\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/emptydir")
set(CPACK_PACKAGE_CONTACT "Acme Maintainer <acme@example.com>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/postinst" "#!/bin/sh\nexit 0\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/conffiles" "/etc/probe.conf\n")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_BINARY_DIR}/postinst;${CMAKE_CURRENT_BINARY_DIR}/conffiles")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION share/probe)
include(CPack)

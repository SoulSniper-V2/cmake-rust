cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" "nested\n")
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION bin COMPONENT runtime)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" DESTINATION lib COMPONENT libraries)
set(CPACK_COMPONENT_RUNTIME_GROUP apps)
set(CPACK_COMPONENT_LIBRARIES_GROUP dev)
set(CPACK_ARCHIVE_FILE_NAME "base")
set(CPACK_ARCHIVE_APPS_FILE_NAME "apps-custom")
set(CPACK_ARCHIVE_DEV_FILE_NAME "dev-custom")
set(CPACK_COMPONENTS_GROUPING ALL_COMPONENTS_IN_ONE)
include(CPack)

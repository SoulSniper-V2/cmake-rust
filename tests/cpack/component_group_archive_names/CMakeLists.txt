cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/a.txt" "a\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/b.txt" "b\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/c.txt" "c\n")
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/a.txt" DESTINATION bin COMPONENT runtime)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/b.txt" DESTINATION lib COMPONENT libraries)
set(CPACK_COMPONENT_RUNTIME_GROUP "apps")
set(CPACK_COMPONENT_LIBRARIES_GROUP "dev")
set(CPACK_COMPONENT_GROUP_APPS_DISPLAY_NAME "Applications")
set(CPACK_COMPONENT_GROUP_DEV_DISPLAY_NAME "Development")
set(CPACK_ARCHIVE_APPS_FILE_NAME "appsgrp")
set(CPACK_ARCHIVE_DEV_FILE_NAME "devgrp")
set(CPACK_COMPONENTS_GROUPING "ONE_PER_GROUP")
include(CPack)

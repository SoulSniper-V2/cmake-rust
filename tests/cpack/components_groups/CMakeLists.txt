cmake_minimum_required(VERSION 3.31)
project(probe NONE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" "hello\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" "nested\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/emptydir")
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hello.txt" DESTINATION bin COMPONENT runtime PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nested.txt" DESTINATION lib COMPONENT libraries)
set(CPACK_COMPONENT_RUNTIME_GROUP "apps")
set(CPACK_COMPONENT_LIBRARIES_GROUP "dev")
set(CPACK_COMPONENT_GROUP_APPS_DISPLAY_NAME "Applications")
set(CPACK_COMPONENT_GROUP_DEV_DISPLAY_NAME "Development")
include(CPack)
